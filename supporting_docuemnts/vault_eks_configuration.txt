# Create service account
kubectl create serviceaccount vault-auth -n default

# Create secret for it (needed in newer k8s versions)
kubectl apply -f - <<EOF
apiVersion: v1
kind: Secret
metadata:
  name: vault-auth-secret
  namespace: default
  annotations:
    kubernetes.io/service-account.name: vault-auth
type: kubernetes.io/service-account-token
EOF

# Get the JWT token
kubectl get secret vault-auth-secret -n default -o jsonpath='{.data.token}' | base64 --decode

1. Policy Name: ai-service-policy
   Content:
   path "secret/data/ai-service" {
  capabilities = ["read"]
}



====================================================
STEP 2 - CREATE AUTH ROLES
====================================================
Go to: Access > Auth Methods > kubernetes > Roles > Create role

5. 
Role Name: ai-service
Alias Name Source: serviceaccount_uid
Audience: https://kubernetes.default.svc
Bound Service Account Names: ai-service, vault-auth
Bound Service Account Namespaces: default
Generated Token Policies: ai-service-policy
Generated Token Type: default
Do Not Attach Default Policy: false
Token Alias Metadata: [object Object]

====================================================
STEP 3 - CREATE SECRETS IN VAULT
====================================================
Go to: Secrets Engines > kv > Create secret

9. Path: ai-service
   Keys to add:
   openai_api_key       = your-openai-api-key
   openai_org_id        = your-openai-org-id





====================================================
STEP 4 - INSTALL VAULT AGENT INJECTOR IN EKS
====================================================
Run these commands from your local machine (kubectl must be pointing to EKS)

13. Add hashicorp helm repo:
    helm repo add hashicorp https://helm.releases.hashicorp.com
    helm repo update

14. Install vault injector only (no vault server):
    helm install vault-agent hashicorp/vault \
      --set "global.externalVaultAddr=https://vault.devopsproduction.com" \
      --set "injector.enabled=true" \
      --set "server.enabled=false" \
      --namespace vault \
      --create-namespace
